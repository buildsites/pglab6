# 13. Image Segmentation

import cv2
from sklearn.cluster import KMeans
import matplotlib.pyplot as plt

image = cv2.imread("image.jpg")
image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
pixels = image.reshape(-1, 3)

model = KMeans(n_clusters=3, n_init=10)
model.fit(pixels)

segmented = model.cluster_centers_[model.labels_]
segmented = segmented.reshape(image.shape).astype("uint8")

plt.subplot(1, 2, 1)
plt.imshow(image)
plt.title("Original")
plt.axis("off")

plt.subplot(1, 2, 2)
plt.imshow(segmented)
plt.title("Segmented")
plt.axis("off")

plt.show()
