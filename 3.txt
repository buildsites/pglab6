# 3. Student Grading System

class Student:
    def __init__(self, name, marks):
        self.name = name
        self.marks = marks

    def display(self):
        print("Name:", self.name)
        print("Marks:", self.marks)

        if self.marks >= 90:
            print("Grade: A")
        elif self.marks >= 75:
            print("Grade: B")
        elif self.marks >= 50:
            print("Grade: C")
        else:
            print("Grade: F")


name = input("Enter Student Name: ")
marks = int(input("Enter Marks: "))

s = Student(name, marks)
s.display()
